cmake_minimum_required(VERSION 3.16)
project(hgvd_exploration)
find_package(ament_cmake REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
foreach(dependency rclcpp rclcpp_action geometry_msgs nav_msgs nav2_msgs tf2 tf2_ros visualization_msgs)
  find_package(${dependency} REQUIRED)
endforeach()

add_library(hgvd_core STATIC src/planner.cpp src/spatial_index.cpp)
target_compile_features(hgvd_core PUBLIC cxx_std_17)
target_include_directories(hgvd_core PUBLIC include PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(hgvd_core PRIVATE ${OpenCV_LIBS})
add_executable(hgvd_explorer src/explorer.cpp)
target_link_libraries(hgvd_explorer hgvd_core)
ament_target_dependencies(hgvd_explorer rclcpp rclcpp_action geometry_msgs nav_msgs nav2_msgs
  tf2 tf2_ros visualization_msgs)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(hgvd_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(hgvd_explorer PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS hgvd_explorer DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(hgvd_core_test test/planner_test.cpp)
  target_link_libraries(hgvd_core_test hgvd_core)
endif()
ament_package()
